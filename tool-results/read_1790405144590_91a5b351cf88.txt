     1→/**
     2→ * StockSense — Inventory Engine (Phase 0/1 core + Phase 3/4 intelligence).
     3→ *
     4→ * Design decisions per the Implementation Plan:
     5→ *  • Split quantities per SKU/location: onHand / reserved / incoming / inTransit / damaged.
     6→ *    available := onHand − reserved (what screens show — nobody can promise units
     7→ *    another order has already claimed).
     8→ *  • Every stock-changing action is a single, uninterruptible check-then-update step
     9→ *    against these states (all mutations run inside one Prisma interactive transaction).
    10→ *  • An immutable, ID'd event ledger entry references the document it came from
    11→ *    (RCPT-…/DEL-…/TRF-…/ADJ-…/CNT-…/OPENING) with previous qty, new qty and the diff.
    12→ *  • Severity rules (Phase 4): LOW <2% auto-logged · MEDIUM 2–15% or repeated mismatch
    13→ *    (flagged for review) · HIGH >15% held for approval · CRITICAL below-zero blocked outright.
    14→ *
    15→ * Every function takes a Prisma client / transaction client as first argument so both
    16→ * API routes (db.$transaction) and the seed script (db) share identical behaviour.
    17→ */
    18→

    29→const DOC_PREFIX: Record<'receipt' | 'deliveryOrder' | 'transfer' | 'adjustment' | 'cycleCount', { prefix: string; start: number }> = {
    30→  receipt: { prefix: 'RCPT-', start: 1000 },
    31→  deliveryOrder: { prefix: 'DEL-', start: 2000 },
    32→  transfer: { prefix: 'TRF-', start: 500 },
    33→  adjustment: { prefix: 'ADJ-', start: 900 },
    34→  cycleCount: { prefix: 'CNT-', start: 40 },
    35→}
    36→
    37→async function nextDocCode(tx: Tx, model: keyof typeof DOC_PREFIX): Promise<string> {
    38→  const { prefix, start } = DOC_PREFIX[model]
    39→  let codes: string[] = []
    40→  if (model === 'receipt') codes = (await tx.receipt.findMany({ select: { code: true } })).map((r) => r.code)
    41→  else if (model === 'deliveryOrder') codes = (await tx.deliveryOrder.findMany({ select: { code: true } })).map((r) => r.code)
    42→  else if (model === 'transfer') codes = (await tx.transfer.findMany({ select: { code: true } })).map((r) => r.code)
    43→  else if (model === 'adjustment') codes = (await tx.adjustment.findMany({ select: { code: true } })).map((r) => r.code)
    44→  else codes = (await tx.cycleCount.findMany({ select: { code: true } })).map((r) => r.code)
    45→  let max = start
    46→  for (const c of codes) {
    47→    const n = parseInt(c.slice(prefix.length), 10)
    48→    if (!isNaN(n) && n > max) max = n
    49→  }
    50→  return `${prefix}${max + 1}`
    51→}
    52→
    53→async function nextLedgerCode(tx: Tx, at: Date = new Date()): Promise<string> {
    54→  const year = at.getUTCFullYear()
    55→  const prefix = `LEDGER-${year}-`
    56→  const rows = await tx.ledgerEntry.findMany({ where: { code: { startsWith: prefix } }, select: { code: true } })
    57→  let max = 0
    58→  for (const r of rows) {
    59→    const n = parseInt(r.code.slice(prefix.length), 10)
    60→    if (!isNaN(n) && n > max) max = n
    61→  }
    62→  return `${prefix}${String(max + 1).padStart(6, '0')}`
    63→}

    90→export async function getStockOrThrow(tx: Tx, productId: number, locationId: number) {

    96→const FIELD_LABEL: Record<keyof StockDeltas, string> = {
    97→  onHand: 'ON_HAND',
    98→  reserved: 'RESERVED',
    99→  incoming: 'INCOMING',
   100→  inTransit: 'IN_TRANSIT',
   101→  damaged: 'DAMAGED',

   104→/**
   105→ * The single place where stock quantities change. Reads current values, applies
   106→ * deltas, refuses anything that would take a quantity below zero (engine-level
   107→ * guarantee), upserts the row and posts one immutable ledger entry per field.
   108→ */
   109→export async function bumpStock(
   110→  tx: Tx,
   111→  productId: number,
   112→  locationId: number,
   113→  deltas: StockDeltas,
   114→  ctx: LedgerCtx
   115→): Promise<{ onHand: number; reserved: number; incoming: number; inTransit: number; damaged: number }> {
   116→  const current = (await getStock(tx, productId, locationId)) ?? { onHand: 0, reserved: 0, incoming: 0, inTransit: 0, damaged: 0 }
   117→  const next = { ...current }
   118→  const applied: { field: keyof StockDeltas; prev: number; newQty: number; diff: number }[] = []
   119→
   120→  for (const field of Object.keys(FIELD_LABEL) as (keyof StockDeltas)[]) {
   121→    const delta = deltas[field] ?? 0
   122→    if (delta === 0) continue
   123→    const prev = next[field]
   124→    const newQty = Math.round((prev + delta) * 1000) / 1000
   125→    if (newQty < 0) {
   126→      throw new HttpError(
   127→        422,
   128→        `Blocked: this change would take ${field} below zero at this location ` +
   129→          `(current ${prev}, applied ${delta >= 0 ? '+' : ''}${delta}). ` +
   130→          `Stock can never be promised twice or go negative.`,
   131→        'CRITICAL'
   132→      )
   133→    }
   134→    next[field] = newQty
   135→    applied.push({ field, prev, newQty, diff: newQty - prev })
   136→  }
   137→
   138→  if (applied.length > 0) {
   139→    await tx.stockLevel.upsert({
   140→      where: { productId_locationId: { productId, locationId } },
   141→      create: { productId, locationId, ...next },
   142→      update: next,
   143→    })
   144→    for (const a of applied) {
   145→      await tx.ledgerEntry.create({
   146→        data: {
   147→          code: await nextLedgerCode(tx, ctx.at),
   148→          docType: ctx.docType,
   149→          docCode: ctx.docCode,
   150→          productId,
   151→          locationId,
   152→          field: FIELD_LABEL[a.field],
   153→          prevQty: a.prev,
   154→          newQty: a.newQty,
   155→          diff: a.diff,
   156→          reason: ctx.reason ?? null,
   157→          performedBy: ctx.performedBy ?? null,
   158→          approvedBy: ctx.approvedBy ?? null,
   159→          ...(ctx.at ? { createdAt: ctx.at } : {}),
   160→        },
   161→      })
   162→    }
   163→  }
   164→  return next
   165→}

   167→async function assertProduct(tx: Tx, productId: number) {
   168→  const p = await tx.product.findUnique({ where: { id: productId } })
   169→  if (!p) throw new HttpError(404, `Unknown product id ${productId}`)
   170→  return p
   171→}

   173→async function assertLocation(tx: Tx, locationId: number) {
   174→  const l = await tx.location.findUnique({ where: { id: locationId }, include: { rack: { include: { zone: true } } } })
   175→  if (!l) throw new HttpError(404, `Unknown location id ${locationId}`)
   176→  return l

   179→/** available := onHand − reserved — the figure every screen must show. */
   180→export function availableOf(s: { onHand: number; reserved: number } | null | undefined): number {
   181→  return (s?.onHand ?? 0) - (s?.reserved ?? 0)
   182→}

   188→export interface ReceiptLineInput {

   194→export async function createReceipt(
   195→  tx: Tx,
   196→  input: { supplierId?: number | null; warehouseId?: number | null; expectedAt?: string; note?: string | null; lines: ReceiptLineInput[] },
   197→  userId: number
   198→) {
   199→  if (!input.lines?.length) throw new HttpError(400, 'A receipt needs at least one line')
   200→  for (const l of input.lines) {
   201→    if (!(l.expectedQty > 0)) throw new HttpError(400, 'Expected quantities must be positive')
   202→    await assertProduct(tx, l.productId)
   203→    await assertLocation(tx, l.locationId)
   204→  }
   205→  const supplier = input.supplierId ? await tx.supplier.findUnique({ where: { id: input.supplierId } }) : null
   206→  if (input.supplierId && !supplier) throw new HttpError(404, `Unknown supplier id ${input.supplierId}`)
   207→  const code = await nextDocCode(tx, 'receipt')
   208→  const receipt = await tx.receipt.create({
   209→    data: {
   210→      code,
   211→      supplierId: supplier?.id ?? null,
   212→      warehouseId: input.warehouseId ?? (await tx.warehouse.findFirst())!.id,
   213→      status: 'EXPECTED',
   214→      expectedAt: input.expectedAt ? new Date(input.expectedAt) : new Date(Date.now() + 3 * 86400000),
   215→      note: input.note ?? null,
   216→      createdBy: userId,
   217→      lines: { create: input.lines.map((l) => ({ productId: l.productId, locationId: l.locationId, expectedQty: l.expectedQty })) },
   218→    },
   219→  })
   220→  for (const l of input.lines) {
   221→    await bumpStock(tx, l.productId, l.locationId, { incoming: l.expectedQty }, {
   222→      docType: 'RECEIPT',
   223→      docCode: code,
   224→      reason: `Receipt ${code} expected${supplier ? ` from ${supplier.name}` : ''}`,
   225→      performedBy: userId,
   226→    })
   227→  }
   228→  return receipt
   229→}

   231→export async function receiveReceipt(
   232→  tx: Tx,
   233→  receiptId: number,
   234→  input: { lines: { lineId: number; receivedQty: number; damagedQty?: number }[]; note?: string | null },
   235→  userId: number
   236→) {
   237→  const receipt = await tx.receipt.findUnique({ where: { id: receiptId }, include: { lines: true, supplier: true } })
   238→  if (!receipt) throw new HttpError(404, 'Receipt not found')
   239→  if (receipt.status !== 'EXPECTED') throw new HttpError(400, `Receipt is ${receipt.status}, only EXPECTED receipts can be received`)
   240→  const byId = new Map(input.lines.map((l) => [l.lineId, l]))
   241→  for (const line of receipt.lines) {
   242→    const given = byId.get(line.id)
   243→    if (!given) throw new HttpError(400, `Missing received quantity for line ${line.id} (${line.productId})`)
   244→    const damaged = Math.max(0, given.damagedQty ?? 0)
   245→    const good = Math.max(0, given.receivedQty) - damaged
   246→    await bumpStock(tx, line.productId, line.locationId, { incoming: -line.expectedQty, onHand: good, ...(damaged > 0 ? { damaged } : {}) }, {
   247→      docType: 'RECEIPT',
   248→      docCode: receipt.code,
   249→      reason: `Receipt ${receipt.code} received${receipt.supplier ? ` from ${receipt.supplier.name}` : ''}` +
   250→        (given.receivedQty !== line.expectedQty ? ` — variance: expected ${line.expectedQty}, received ${given.receivedQty}` : ''),
   251→      performedBy: userId,
   252→    })
   253→    await tx.receiptLine.update({ where: { id: line.id }, data: { receivedQty: Math.max(0, given.receivedQty) } })
   254→  }
   255→  return tx.receipt.update({
   256→    where: { id: receiptId },
   257→    data: { status: 'RECEIVED', receivedAt: new Date(), note: input.note ?? receipt.note },
   258→  })
   259→}

   261→export async function cancelReceipt(tx: Tx, receiptId: number, userId: number) {
   262→  const receipt = await tx.receipt.findUnique({ where: { id: receiptId }, include: { lines: true } })
   263→  if (!receipt) throw new HttpError(404, 'Receipt not found')
   264→  if (receipt.status !== 'EXPECTED') throw new HttpError(400, 'Only EXPECTED receipts can be cancelled')
   265→  for (const line of receipt.lines) {
   266→    await bumpStock(tx, line.productId, line.locationId, { incoming: -line.expectedQty }, {
   267→      docType: 'RECEIPT',
   268→      docCode: receipt.code,
   269→      reason: `Receipt ${receipt.code} cancelled — incoming expectation released`,
   270→      performedBy: userId,
   271→    })
   272→  }
   273→  return tx.receipt.update({ where: { id: receiptId }, data: { status: 'CANCELLED' } })
   274→}

   276→// ---------------------------------------------------------------------------
   277→// Delivery orders — Available → Reserved → Picked → Packed → Delivered
   278→// Stock is reserved the moment the order is created and only deducted once packed:
   279→// "order created" and "stock physically left the warehouse" are never the same thing.
   280→// ---------------------------------------------------------------------------
   281→
   282→export async function createDelivery(
   283→  tx: Tx,
   284→  input: { customer: string; note?: string | null; lines: { productId: number; locationId: number; qty: number }[] },
   285→  userId: number
   286→) {
   287→  if (!input.customer?.trim()) throw new HttpError(400, 'Customer name is required')
   288→  if (!input.lines?.length) throw new HttpError(400, 'A delivery order needs at least one line')
   289→  for (const l of input.lines) {
   290→    await assertProduct(tx, l.productId)
   291→    await assertLocation(tx, l.locationId)
   292→    if (!(l.qty > 0)) throw new HttpError(400, 'Quantities must be positive')
   293→    const stock = await getStock(tx, l.productId, l.locationId)
   294→    const available = availableOf(stock)
   295→    if (l.qty > available) {
   296→      const p = await tx.product.findUnique({ where: { id: l.productId } })
   297→      throw new HttpError(
   298→        400,
   299→        `Insufficient available stock for ${p?.sku} at this location: available ${available}, requested ${l.qty}. ` +
   300→          `Units already reserved by other orders can never be promised twice.`
   301→      )
   302→    }
   303→  }
   304→  const code = await nextDocCode(tx, 'deliveryOrder')
   305→  const order = await tx.deliveryOrder.create({
   306→    data: {
   307→      code,
   308→      customer: input.customer.trim(),
   309→      status: 'RESERVED',
   310→      note: input.note ?? null,
   311→      createdBy: userId,
   312→      lines: { create: input.lines.map((l) => ({ productId: l.productId, locationId: l.locationId, qty: l.qty })) },
   313→    },
   314→  })
   315→  for (const l of input.lines) {
   316→    await bumpStock(tx, l.productId, l.locationId, { reserved: l.qty }, {
   317→      docType: 'DELIVERY',
   318→      docCode: code,
   319→      reason: `Delivery ${code} created for ${input.customer.trim()} — stock reserved`,
   320→      performedBy: userId,
   321→    })
   322→  }
   323→  return order
   324→}

   330→  return tx.deliveryOrder.update({ where: { id: orderId }, data: { status: 'PICKED', pickedAt: new Date() } })

   333→export async function markDeliveryPacked(tx: Tx, orderId: number, userId: number) {
   334→  const order = await tx.deliveryOrder.findUnique({ where: { id: orderId }, include: { lines: true } })
   335→  if (!order) throw new HttpError(404, 'Delivery order not found')
   336→  if (order.status !== 'RESERVED' && order.status !== 'PICKED') {
   337→    throw new HttpError(400, `Order is ${order.status}; only RESERVED or PICKED orders can be marked packed`)
   338→  }
   339→  for (const line of order.lines) {
   340→    await bumpStock(tx, line.productId, line.locationId, { onHand: -line.qty, reserved: -line.qty }, {
   341→      docType: 'DELIVERY',
   342→      docCode: order.code,
   343→      reason: `Delivery ${order.code} packed for ${order.customer} — stock physically left the warehouse`,
   344→      performedBy: userId,
   345→    })
   346→    await tx.deliveryLine.update({ where: { id: line.id }, data: { pickedQty: line.qty } })
   347→  }
   348→  return tx.deliveryOrder.update({ where: { id: orderId }, data: { status: 'PACKED', packedAt: new Date() } })
   349→}

   379→export async function createTransfer(
   380→  tx: Tx,
   381→  input: { fromLocationId: number; toLocationId: number; note?: string | null; lines: { productId: number; qty: number }[] },
   382→  userId: number
   383→) {
   384→  if (input.fromLocationId === input.toLocationId) throw new HttpError(400, 'Source and destination must differ')
   385→  if (!input.lines?.length) throw new HttpError(400, 'A transfer needs at least one line')
   386→  for (const l of input.lines) {
   387→    await assertProduct(tx, l.productId)
   388→    if (!(l.qty > 0)) throw new HttpError(400, 'Quantities must be positive')
   389→    const stock = await getStock(tx, l.productId, input.fromLocationId)
   390→    const available = availableOf(stock)
   391→    if (l.qty > available) {
   392→      const p = await tx.product.findUnique({ where: { id: l.productId } })
   393→      throw new HttpError(400, `Insufficient available stock for ${p?.sku} at source: available ${available}, requested ${l.qty}`)
   394→    }
   395→  }
   396→  await assertLocation(tx, input.fromLocationId)
   397→  await assertLocation(tx, input.toLocationId)
   398→  const code = await nextDocCode(tx, 'transfer')
   399→  const transfer = await tx.transfer.create({
   400→    data: {
   401→      code,
   402→      status: 'IN_TRANSIT',
   403→      fromLocationId: input.fromLocationId,
   404→      toLocationId: input.toLocationId,
   405→      note: input.note ?? null,
   406→      createdBy: userId,
   407→      shippedAt: new Date(),
   408→      lines: { create: input.lines.map((l) => ({ productId: l.productId, qty: l.qty })) },
   409→    },
   410→  })
   411→  for (const l of input.lines) {
   412→    await bumpStock(tx, l.productId, input.fromLocationId, { onHand: -l.qty }, {
   413→      docType: 'TRANSFER',
   414→      docCode: code,
   415→      reason: `Transfer ${code} shipped — left source`,
   416→      performedBy: userId,
   417→    })
   418→    await bumpStock(tx, l.productId, input.toLocationId, { inTransit: l.qty }, {
   419→      docType: 'TRANSFER',
   420→      docCode: code,
   421→      reason: `Transfer ${code} in transit — arriving`,
   422→      performedBy: userId,
   423→    })
   424→  }
   425→  return transfer
   426→}

   428→export async function receiveTransfer(tx: Tx, transferId: number, userId: number) {

   432→  for (const line of transfer.lines) {

   435→      docCode: transfer.code,
   436→      reason: `Transfer ${transfer.code} received — destination available`,
   437→      performedBy: userId,
   438→    })
   439→  }
   440→  return tx.transfer.update({ where: { id: transferId }, data: { status: 'RECEIVED', receivedAt: new Date() } })
   441→}

   465→// Stock adjustments + severity rules (Phase 4 governance)
   466→//   LOW      < 2% one-off mismatch        → posted, logged automatically
   467→//   MEDIUM   2–15%, or repeated mismatch  → posted, flagged for manager review
   468→//   HIGH     > 15% of stock               → held for manager approval before posting
   469→//   CRITICAL would go below zero          → blocked outright (never stored)
   470→// ---------------------------------------------------------------------------
   471→
   472→export const SEVERITY = { LOW_PCT: 0.02, HIGH_PCT: 0.15, REPEATED_DAYS: 30, REPEATED_PRIORS: 2, STAFF_7D: 6 }
   473→
   474→export interface AdjustLineInput {
   475→  productId: number
   476→  locationId: number
   477→  countedQty: number
   478→}
   479→
   480→export interface AdjustOutcome {
   481→  adjustment: { id: number; code: string; status: string; severity: string; reason: string; note: string | null; createdAt: Date; postedAt: Date | null }
   482→  explanation: string
   483→  flagsCreated: { type: string; message: string }[]
   484→}
   485→
   486→export async function createAdjustment(
   487→  tx: Tx,
   488→  input: { reason: string; note?: string | null; lines: AdjustLineInput[]; sourceCountCode?: string },
   489→  userId: number
   490→): Promise<AdjustOutcome> {
   491→  if (!input.reason?.trim()) throw new HttpError(400, 'A reason is required for every stock adjustment')
   492→  if (!input.lines?.length) throw new HttpError(400, 'An adjustment needs at least one line')
   493→
   494→  const prepared: { productId: number; locationId: number; systemQty: number; countedQty: number; delta: number; pct: number; downward: boolean }[] = []
   495→  for (const l of input.lines) {
   496→    const product = await assertProduct(tx, l.productId)
   497→    await assertLocation(tx, l.locationId)
   498→    const stock = await getStock(tx, l.productId, l.locationId)
   499→    const systemQty = stock?.onHand ?? 0
   500→    if (l.countedQty < 0) {
   501→      throw new HttpError(
   502→        422,
   503→        `Blocked (CRITICAL): entering ${l.countedQty} ${product.unit} for ${product.sku} would take stock below zero. Adjustments can never set negative stock.`,
   504→        'CRITICAL'
   505→      )
   506→    }
   507→    const delta = Math.round((l.countedQty - systemQty) * 1000) / 1000
   508→    const pct = systemQty > 0 ? Math.abs(delta) / systemQty : delta > 0 ? 1 : 0
   509→    prepared.push({ productId: l.productId, locationId: l.locationId, systemQty, countedQty: l.countedQty, delta, pct, downward: delta < 0 })
   510→  }
   511→
   512→  // severity classification
   513→  let severity: 'LOW' | 'MEDIUM' | 'HIGH' = 'LOW'
   514→  const big = prepared.filter((p) => p.downward && p.pct > SEVERITY.HIGH_PCT)
   515→  const moderate = prepared.filter((p) => p.downward && p.pct >= SEVERITY.LOW_PCT && p.pct <= SEVERITY.HIGH_PCT)
   516→  if (big.length > 0) severity = 'HIGH'
   517→  else if (moderate.length > 0) severity = 'MEDIUM'
   518→
   519→  // repeated-mismatch pattern: ≥3rd downward adjustment at same product+location in 30 days
   520→  const flags: { type: string; message: string }[] = []
   521→  const since = new Date(Date.now() - SEVERITY.REPEATED_DAYS * 86400000)
   522→  const repeatedAt: typeof prepared = []
   523→  for (const p of prepared.filter((x) => x.downward)) {
   524→    const priors = await tx.adjustmentLine.count({
   525→      where: {
   526→        productId: p.productId,
   527→        locationId: p.locationId,
   528→        delta: { lt: 0 },
   529→        adjustment: { createdAt: { gte: since }, status: { in: ['POSTED', 'PENDING_APPROVAL'] } },
   530→      },
   531→    })
   532→    if (priors >= SEVERITY.REPEATED_PRIORS) {
   533→      repeatedAt.push(p)
   534→      if (severity === 'LOW') severity = 'MEDIUM'
   535→    }
   536→  }
   537→
   538→  // staff anomaly: unusually high adjustment count by one user in 7 days (review item, not an accusation)
   539→  const staffAdjustments7d = await tx.adjustment.count({
   540→    where: { createdBy: userId, createdAt: { gte: new Date(Date.now() - 7 * 86400000) } },
   541→  })
   542→
   543→  const code = await nextDocCode(tx, 'adjustment')
   544→  const reasonText = input.sourceCountCode ? `${input.reason} (cycle count ${input.sourceCountCode})` : input.reason.trim()
   545→
   546→  const adjustment = await tx.adjustment.create({
   547→    data: {
   548→      code,
   549→      status: severity === 'HIGH' ? 'PENDING_APPROVAL' : 'POSTED',
   550→      severity,
   551→      reason: reasonText,
   552→      note: input.note ?? null,
   553→      createdBy: userId,
   554→      postedAt: severity === 'HIGH' ? null : new Date(),
   555→      lines: {
   556→        create: prepared.map((p) => ({
   557→          productId: p.productId,
   558→          locationId: p.locationId,
   559→          systemQty: p.systemQty,
   560→          countedQty: p.countedQty,
   561→          delta: p.delta,
   562→        })),
   563→      },
   564→    },
   565→  })
   566→
   567→  if (severity !== 'HIGH') {
   568→    for (const p of prepared) {
   569→      if (p.delta === 0) continue
   570→      await bumpStock(tx, p.productId, p.locationId, { onHand: p.delta }, {
   571→        docType: input.sourceCountCode ? 'COUNT' : 'ADJUSTMENT',
   572→        docCode: code,
   573→        reason: reasonText,
   574→        performedBy: userId,
   575→      })
   576→    }
   577→  }
   578→
   579→  // plain-language explanations, mirroring the plan's example style
   580→  const parts: string[] = []
   581→  if (severity === 'HIGH') {
   582→    parts.push(
   583→      `Held for manager approval: removes ${Math.round(big[0].pct * 100)}% of current stock at this location` +
   584→        (repeatedAt.length > 0 ? ` and is a repeated downward adjustment (3rd+ in ${SEVERITY.REPEATED_DAYS} days)` : '') +
   585→        `. It will only post once approved.`
   586→    )
   587→  } else if (repeatedAt.length > 0) {
   588→    parts.push(`Flagged for review: this is the 3rd+ downward adjustment at this product/location within ${SEVERITY.REPEATED_DAYS} days.`)
   589→  } else if (severity === 'MEDIUM') {
   590→    parts.push(`Flagged for review: adjusts ${Math.round(moderate[0].pct * 100)}% of stock at this location.`)
   591→  } else {
   592→    parts.push(`Logged automatically: one-off mismatch under ${Math.round(SEVERITY.LOW_PCT * 100)}% of stock — no action needed.`)
   593→  }
   594→
   595→  if (repeatedAt.length > 0) {
   596→    flags.push({
   597→      type: 'REPEATED_MISMATCH',
   598→      message: `Repeated mismatch: 3rd+ downward adjustment within ${SEVERITY.REPEATED_DAYS} days at the same product/location (${repeatedAt.map((r) => `#${r.productId}`).join(', ')}).`,
   599→    })
   600→  } else if (severity === 'MEDIUM') {
   601→    flags.push({
   602→      type: 'MODERATE_VARIANCE',
   603→      message: `Moderate variance: adjusts ${Math.round(moderate[0].pct * 100)}% of stock at this location — flagged for manager review.`,
   604→    })
   605→  }
   606→  if (staffAdjustments7d + 1 > SEVERITY.STAFF_7D) {
   607→    flags.push({
   608→      type: 'STAFF_ANOMALY',
   609→      message: `Unusually high adjustment activity: ${staffAdjustments7d + 1} adjustments by this user in the past 7 days — surfaced as a review item.`,
   610→    })
   611→  }
   612→  for (const f of flags) {
   613→    await tx.exceptionFlag.create({
   614→      data: { type: f.type, severity: 'MEDIUM', message: f.message, refType: 'ADJUSTMENT', refId: adjustment.id, refCode: code, status: 'OPEN', createdBy: userId },
   615→    })
   616→  }
   617→
   618→  return { adjustment, explanation: parts.join(' '), flagsCreated: flags }
   619→}

   621→export async function approveAdjustment(tx: Tx, adjustmentId: number, approverId: number) {
   622→  const adjustment = await tx.adjustment.findUnique({ where: { id: adjustmentId }, include: { lines: true } })
   623→  if (!adjustment) throw new HttpError(404, 'Adjustment not found')
   624→  if (adjustment.status !== 'PENDING_APPROVAL') throw new HttpError(400, `Adjustment is ${adjustment.status}; only PENDING_APPROVAL adjustments can be approved`)
   625→  for (const line of adjustment.lines) {
   626→    if (line.delta === 0) continue
   627→    await bumpStock(tx, line.productId, line.locationId, { onHand: line.delta }, {
   628→      docType: 'ADJUSTMENT',
   629→      docCode: adjustment.code,
   630→      reason: adjustment.reason,
   631→      performedBy: adjustment.createdBy ?? null,
   632→      approvedBy: approverId,
   633→    })
   634→  }
   635→  return tx.adjustment.update({ where: { id: adjustmentId }, data: { status: 'POSTED', approvedBy: approverId, postedAt: new Date() } })
   636→}

   646→// Cycle counts (Phase 4) — only a real variance opens an adjustment

   649→const CADENCE_BY_CLASS: Record<string, string> = { HIGH: 'WEEKLY', MEDIUM: 'MONTHLY', LOW: 'QUARTERLY' }
   650→const CADENCE_DAYS: Record<string, number> = { WEEKLY: 7, MONTHLY: 30, QUARTERLY: 91 }
   651→
   652→export async function createCount(
   653→  tx: Tx,
   654→  input: { scope: 'LOCATION' | 'PRODUCT'; locationId?: number; productId?: number; dueDate?: string; note?: string | null },
   655→  userId: number
   656→) {
   657→  let lines: { productId: number; systemQty: number }[] = []
   658→  let cadence = 'MONTHLY'
   659→  if (input.scope === 'LOCATION') {
   660→    if (!input.locationId) throw new HttpError(400, 'locationId required for a LOCATION count')
   661→    await assertLocation(tx, input.locationId)
   662→    const stocks = await tx.stockLevel.findMany({ where: { locationId: input.locationId }, include: { product: true } })
   663→    lines = stocks.map((s) => ({ productId: s.productId, systemQty: s.onHand }))
   664→    const classes = new Set(stocks.map((s) => s.product.valueClass))
   665→    cadence = classes.has('HIGH') ? 'WEEKLY' : classes.has('MEDIUM') ? 'MONTHLY' : 'QUARTERLY'
   666→  } else {
   667→    if (!input.productId) throw new HttpError(400, 'productId required for a PRODUCT count')
   668→    const product = await assertProduct(tx, input.productId)
   669→    const stocks = await tx.stockLevel.findMany({ where: { productId: input.productId } })
   670→    lines = stocks.map((s) => ({ productId: s.productId, systemQty: s.onHand }))
   671→    cadence = CADENCE_BY_CLASS[product.valueClass] ?? 'MONTHLY'
   672→  }
   673→  if (lines.length === 0) throw new HttpError(400, 'Nothing to count at this scope (no stock records found)')
   674→  const code = await nextDocCode(tx, 'cycleCount')
   675→  const due = input.dueDate ? new Date(input.dueDate) : new Date(Date.now() + (CADENCE_DAYS[cadence] ?? 30) * 86400000)
   676→  const count = await tx.cycleCount.create({
   677→    data: {
   678→      code,
   679→      status: 'OPEN',
   680→      scope: input.scope,
   681→      locationId: input.scope === 'LOCATION' ? input.locationId! : null,
   682→      productId: input.scope === 'PRODUCT' ? input.productId! : null,
   683→      dueDate: due,
   684→      cadence,
   685→      note: input.note ?? null,
   686→      lines: { create: lines },
   687→    },
   688→  })
   689→  return count
   690→}

   692→export async function submitCount(
   693→  tx: Tx,
   694→  countId: number,
   695→  input: { lines: { lineId: number; countedQty: number }[]; note?: string | null },
   696→  userId: number
   697→) {
   698→  const count = await tx.cycleCount.findUnique({ where: { id: countId }, include: { lines: { include: { product: true } } } })
   699→  if (!count) throw new HttpError(404, 'Cycle count not found')
   700→  if (count.status !== 'OPEN') throw new HttpError(400, `Count is ${count.status}; only OPEN counts can be submitted`)
   701→  const byId = new Map(input.lines.map((l) => [l.lineId, l]))
   702→  const varianceLines: { productId: number; locationId: number; countedQty: number }[] = []
   703→  const locationId = count.locationId
   704→  for (const line of count.lines) {
   705→    const given = byId.get(line.id)
   706→    if (given == null) throw new HttpError(400, `Missing counted quantity for ${line.product.sku}`)
   707→    const variance = Math.round((given.countedQty - line.systemQty) * 1000) / 1000
   708→    await tx.cycleCountLine.update({ where: { id: line.id }, data: { countedQty: given.countedQty, variance } })
   709→    if (variance !== 0) {
   710→      const locId = locationId ?? (await tx.stockLevel.findFirst({ where: { productId: line.productId } }))?.locationId
   711→      if (locId) varianceLines.push({ productId: line.productId, locationId: locId, countedQty: given.countedQty })
   712→    }
   713→  }
   714→  const updated = await tx.cycleCount.update({
   715→    where: { id: countId },
   716→    data: { status: 'COMPLETED', completedAt: new Date(), completedBy: userId, note: input.note ?? count.note },
   717→  })
   718→  let outcome: AdjustOutcome | null = null
   719→  if (varianceLines.length > 0) {
   720→    // only a real variance opens an adjustment — and it passes through the same severity engine
   721→    outcome = await createAdjustment(
   722→      tx,
   723→      { reason: `Cycle count ${count.code} variance on ${varianceLines.length} SKU(s)`, lines: varianceLines, sourceCountCode: count.code },
   724→      userId
   725→    )
   726→  }
   727→  return { count: updated, outcome }
   728→}

   737→// ---------------------------------------------------------------------------
   738→// Reorder intelligence (Phase 3) — supplier-aware, explainable
   739→// projected available = on-hand + incoming − reserved  (vs the reorder point)
   740→// ---------------------------------------------------------------------------

   742→export interface ProductNeed {
   743→  productId: number
   744→  sku: string
   745→  name: string
   746→  unit: string
   747→  category: string
   748→  onHand: number
   749→  reserved: number
   750→  incoming: number
   751→  inTransit: number
   752→  projectedAvailable: number
   753→  reorderPoint: number
   754→  dailyUsage: number
   755→  safetyStock: number
   756→  belowReorder: boolean
   757→  stockoutRisk: boolean
   758→}
   759→
   760→export async function computeNeeds(tx: Tx = db): Promise<ProductNeed[]> {
   761→  const products = await tx.product.findMany({
   762→    where: { active: true },
   763→    include: { stocks: true, suppliers: { include: { supplier: true } } },
   764→    orderBy: { sku: 'asc' },
   765→  })
   766→  return products.map((p) => {
   767→    const onHand = p.stocks.reduce((a, s) => a + s.onHand, 0)
   768→    const reserved = p.stocks.reduce((a, s) => a + s.reserved, 0)
   769→    const incoming = p.stocks.reduce((a, s) => a + s.incoming, 0)
   770→    const inTransit = p.stocks.reduce((a, s) => a + s.inTransit, 0)
   771→    const projectedAvailable = Math.round((onHand + incoming - reserved) * 1000) / 1000
   772→    return {
   773→      productId: p.id,
   774→      sku: p.sku,
   775→      name: p.name,
   776→      unit: p.unit,
   777→      category: p.category,
   778→      onHand,
   779→      reserved,
   780→      incoming,
   781→      inTransit,
   782→      projectedAvailable,
   783→      reorderPoint: p.reorderPoint,
   784→      dailyUsage: p.dailyUsage,
   785→      safetyStock: p.safetyStock,
   786→      belowReorder: projectedAvailable < p.reorderPoint,
   787→      stockoutRisk: projectedAvailable <= p.safetyStock,
   788→    }
   789→  })
   790→}

   792→export function buildReorderReason(need: ProductNeed, supplierName: string | null, leadDays: number, suggestedQty: number, minOrderQty: number, orderMultiple: number): string {
   793→  const u = need.unit
   794→  const lead = supplierName ? `${supplierName}'s ${leadDays}-day lead time` : `a ${leadDays}-day lead time`
   795→  const moq = minOrderQty > 0 ? ` (MOQ ${minOrderQty} ${u}, order multiple ${orderMultiple} ${u})` : orderMultiple > 1 ? ` (order multiple ${orderMultiple} ${u})` : ''
   796→  return (
   797→    `On-hand: ${need.onHand} ${u} · Reserved: ${need.reserved} ${u} · Incoming: ${need.incoming} ${u} · Available: ${need.projectedAvailable} ${u}\n` +
   798→    `Reorder point: ${need.reorderPoint} ${u} (${need.dailyUsage} ${u}/day × ${leadDays}-day lead time + ${need.safetyStock} ${u} safety stock)\n` +
   799→    `→ Reorder recommended: projected available stock (${need.projectedAvailable} ${u}) is below the ${need.reorderPoint} ${u} reorder point within ${lead}.\n` +
   800→    `→ Suggested quantity: ${suggestedQty} ${u}${moq}` +
   801→    (supplierName ? ` · Preferred supplier: ${supplierName}` : '')
   802→  )
   803→}

   805→/** Recompute PENDING suggestions for every product needing reorder (idempotent). */
   806→export async function refreshSuggestions(tx: Tx = db) {
   807→  const needs = await computeNeeds(tx)
   808→  const needing = needs.filter((n) => n.belowReorder)
   809→  const withSuppliers = await tx.product.findMany({
   810→    where: { id: { in: needing.map((n) => n.productId) } },
   811→    include: { suppliers: { include: { supplier: true } } },
   812→  })
   813→  const byId = new Map(withSuppliers.map((p) => [p.id, p]))
   814→  const now = new Date()
   815→
   816→  for (const need of needing) {
   817→    const existing = await tx.reorderSuggestion.findUnique({ where: { productId: need.productId } })
   818→    if (existing) {
   819→      if (existing.status === 'ACCEPTED') continue // already converted to an expected receipt
   820→      if (existing.status === 'DISMISSED' && existing.decidedAt && now.getTime() - existing.decidedAt.getTime() < 7 * 86400000) continue
   821→    }
   822→    const product = byId.get(need.productId)
   823→    const links = (product?.suppliers ?? []).slice().sort(
   824→      (a, b) => Number(b.preferred) - Number(a.preferred) || a.supplier.leadTimeDays - b.supplier.leadTimeDays
   825→    )
   826→    const link = links[0]
   827→    const leadDays = link?.supplier.leadTimeDays ?? 7
   828→    const moq = link?.minOrderQty ?? 0
   829→    const multiple = link?.orderMultiple && link.orderMultiple > 0 ? link.orderMultiple : 1
   830→    const target = need.dailyUsage * (leadDays + 7) + need.safetyStock // cover lead + review period + safety
   831→    let qty = Math.max(0, target - need.projectedAvailable)
   832→    qty = Math.ceil(qty / multiple) * multiple
   833→    if (moq > 0) qty = Math.max(qty, Math.ceil(moq / multiple) * multiple)
   834→    qty = Math.max(qty, multiple)
   835→    const reason = buildReorderReason(need, link?.supplier.name ?? null, leadDays, qty, moq, multiple)
   836→    const data = {
   837→      suggestedQty: qty,
   838→      preferredName: link?.supplier.name ?? '—',
   839→      preferredLeadDays: leadDays,
   840→      reason,
   841→      status: 'PENDING',
   842→      decidedBy: null,
   843→      decidedAt: null,
   844→      receiptCode: null,
   845→    }
   846→    await tx.reorderSuggestion.upsert({ where: { productId: need.productId }, create: { productId: need.productId, ...data }, update: data })
   847→  }
   848→  return tx.reorderSuggestion.findMany({ include: { product: true }, orderBy: { updatedAt: 'desc' } })
   849→}

   851→export async function acceptSuggestion(tx: Tx, suggestionId: number, userId: number, expectedAt?: string) {
   852→  const s = await tx.reorderSuggestion.findUnique({ where: { id: suggestionId }, include: { product: { include: { stocks: true } } } })
   853→  if (!s) throw new HttpError(404, 'Suggestion not found')
   854→  if (s.status !== 'PENDING') throw new HttpError(400, `Suggestion is ${s.status}; only PENDING suggestions can be accepted`)
   855→  const product = s.product
   856→  const stockRows = product.stocks.slice().sort((a, b) => b.onHand - a.onHand)
   857→  const locationId = stockRows[0]?.locationId ?? (await tx.location.findFirst())!.id
   858→  const supplier = await tx.supplier.findUnique({ where: { name: s.preferredName } })
   859→  const receipt = await createReceipt(
   860→    tx,
   861→    {
   862→      supplierId: supplier?.id ?? null,
   863→      expectedAt: expectedAt ?? new Date(Date.now() + s.preferredLeadDays * 86400000).toISOString(),
   864→      note: `Created from reorder suggestion for ${product.sku}`,
   865→      lines: [{ productId: product.id, locationId, expectedQty: s.suggestedQty }],
   866→    },
   867→    userId
   868→  )
   869→  return tx.reorderSuggestion.update({
   870→    where: { id: suggestionId },
   871→    data: { status: 'ACCEPTED', decidedBy: userId, decidedAt: new Date(), receiptCode: receipt.code },
   872→  })
   873→}
